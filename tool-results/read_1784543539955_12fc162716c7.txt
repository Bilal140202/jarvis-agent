    83→  Future<void> _checkPermissions() async {
    84→    final accessibilityRunning = await _screenAutomationService
    85→        .isServiceRunning();
    86→    final microphoneStatus = await Permission.microphone.status;
    87→    final notificationsStatus = await Permission.notification.status;
    88→    final contactsStatus = await Permission.contacts.status;
    89→    final phoneStatus = await Permission.phone.status;
    90→    final smsStatus = await Permission.sms.status;
    91→    final overlayGranted = FeatureFlags.floatingOverlayEnabled
    92→        ? await FlutterOverlayWindow.isPermissionGranted()
    93→        : false;
    94→
    95→    if (mounted) {
    96→      setState(() {
    97→        _isAccessibilityGranted = accessibilityRunning;
    98→        _isMicrophoneGranted = microphoneStatus.isGranted;
    99→        _isNotificationsGranted = notificationsStatus.isGranted;
   100→        _isContactsGranted = contactsStatus.isGranted;
   101→        _isPhoneGranted = phoneStatus.isGranted;
   102→        _isSmsGranted = smsStatus.isGranted;
   103→        _isOverlayGranted = overlayGranted;
   104→      });
   105→    }
   106→  }

   182→  Future<void> _testAndSave() async {
   183→    setState(() {
   184→      _isValidating = true;
   185→      _validationError = null;
   186→    });
   187→
   188→    final apiKey = _apiKeyController.text.trim();
   189→    final baseUrl = _baseUrlController.text.trim();
   190→    final model = _modelController.text.trim();
   191→
   192→    if (baseUrl.isEmpty || model.isEmpty) {
   193→      setState(() {
   194→        _validationError = 'Please fill out API Base URL and Model.';
   195→        _isValidating = false;
   196→      });
   197→      return;
   198→    }
   199→
   200→    if (_selectedProvider != 'ollama' &&
   201→        _selectedProvider != 'local' &&
   202→        apiKey.isEmpty) {
   203→      setState(() {
   204→        _validationError = 'API Key is required for this provider.';
   205→        _isValidating = false;
   206→      });
   207→      return;
   208→    }
   209→
   210→    try {
   211→      final models = await _aiService.fetchAvailableModels(baseUrl, apiKey);
   212→      if (models.isNotEmpty ||
   213→          _selectedProvider == 'ollama' ||
   214→          _selectedProvider == 'local') {
   215→        await _aiService.saveSettings(
   216→          apiKey: apiKey,
   217→          baseUrl: baseUrl,
   218→          model: model,
   219→        );
   220→        final prefs = await SharedPreferences.getInstance();
   221→        await prefs.setBool('onboarding_completed', true);
   222→
   223→        if (mounted) {
   224→          setState(() {
   225→            _isValidating = false;
   226→          });
   227→
   228→          ScaffoldMessenger.of(context).showSnackBar(
   229→            SnackBar(
   230→              content: const Text(
   231→                'Configuration validated! Launching PrivateAgent...',
   232→              ),
   233→              backgroundColor: Colors.indigoAccent,
   234→              behavior: SnackBarBehavior.floating,
   235→              shape: RoundedRectangleBorder(
   236→                borderRadius: BorderRadius.circular(12),
   237→              ),
   238→            ),
   239→          );
   240→
   241→          Navigator.pushReplacement(
   242→            context,
   243→            MaterialPageRoute(builder: (_) => const HomeScreen()),
   244→          );
   245→        }
   246→      } else {
   247→        setState(() {
   248→          _validationError =
   249→              'Failed to fetch models from the server. Verify base URL and API Key.';
   250→          _isValidating = false;
   251→        });
   252→      }
   253→    } catch (e) {
   254→      setState(() {
   255→        _validationError =
   256→            'Error: ${e.toString().replaceFirst('Exception: ', '')}';
   257→        _isValidating = false;
   258→      });
   259→    }
   260→  }
