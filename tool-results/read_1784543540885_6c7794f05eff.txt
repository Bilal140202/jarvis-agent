    73→  Future<void> _saveSession() async {
    74→    if (_messages.isEmpty) return;
    75→
    76→    // Set first user message as session title if not set
    77→    if (_sessionTitle.isEmpty) {
    78→      final firstUserMsg = _messages.firstWhere(
    79→        (m) => m.isUser,
    80→        orElse: () => ChatMessage(role: 'user', content: 'New Chat'),
    81→      );
    82→      _sessionTitle = firstUserMsg.content.length > 28
    83→          ? '${firstUserMsg.content.substring(0, 25)}...'
    84→          : firstUserMsg.content;
    85→    }
    86→
    87→    final session = ChatSession(
    88→      id: _sessionId,
    89→      title: _sessionTitle,
    90→      timestamp: DateTime.now(),
    91→      messages: _messages.map((m) => m.toJson()).toList(),
    92→    );
    93→
    94→    await ChatHistoryService.saveSession(session);
    95→  }

    97→  Future<void> _sendMessage(String text) async {
    98→    if (text.trim().isEmpty) return;
    99→
   100→    final userMessage = ChatMessage(role: 'user', content: text.trim());
   101→    setState(() {
   102→      _messages.add(userMessage);
   103→      _isLoading = true;
   104→    });
   105→    _updateOverlayState();
   106→    _textController.clear();
   107→    _scrollToBottom();
   108→    await _saveSession();
   109→
   110→    // Add empty placeholder assistant message for streaming
   111→    final assistantMessage = ChatMessage(role: 'assistant', content: '');
   112→    setState(() {
   113→      _messages.add(assistantMessage);
   114→    });
   115→    final assistantIndex = _messages.length - 1;
   116→
   117→    try {
   118→      final isAgent = _mode == 'agent';
   119→      final stream = _aiService
   120→          .sendMessageStream(text.trim(), isAgentMode: isAgent)
   121→          .timeout(
   122→            const Duration(seconds: 90),
   123→            onTimeout: (sink) {
   124→              sink.addError(
   125→                TimeoutException(
   126→                  'The model did not return visible text within 90 seconds.',
   127→                ),
   128→              );
   129→              sink.close();
   130→            },
   131→          );
   132→      String accumulated = '';
   133→
   134→      await for (final chunk in stream) {
   135→        accumulated += chunk;
   136→        if (mounted) {
   137→          setState(() {
   138→            _messages[assistantIndex] = ChatMessage(
   139→              role: 'assistant',
   140→              content: accumulated,
   141→            );
   142→          });
   143→          _scrollToBottom();
   144→        }
   145→      }
   146→      await _saveSession();
   147→
   148→      // Check if it's an action
   149→      final action = _aiService.parseAction(accumulated);
   150→
   151→      if (action != null) {
   152→        // If it's an action, we remove the raw JSON message from display
   153→        setState(() {
   154→          _messages.removeAt(assistantIndex);
   155→        });
   156→
   157→        await _showTaskProgressOverlay('Starting: ${text.trim()}');
   158→
   159→        // Execute the action (pass aiService for multi-step tasks)
   160→        final result = await _actionHandler.execute(
   161→          action,
   162→          aiService: _aiService,
   163→          onProgress: (msg) {
   164→            developer.log('Task progress: $msg', name: 'PrivateAgent');
   165→            _sendOverlayEvent('OVERLAY_PROGRESS', msg);
   166→            if (mounted) {
   167→              setState(() {
   168→                _messages.add(
   169→                  ChatMessage(role: 'assistant', content: '⏳ $msg'),
   170→                );
   171→              });
   172→              _scrollToBottom();
   173→            }
   174→          },
   175→        );
   176→
   177→        setState(() {
   178→          _messages.add(
   179→            ChatMessage(
   180→              role: 'assistant',
   181→              content: result.success
   182→                  ? (action.response.isNotEmpty
   183→                        ? action.response
   184→                        : (result.details ?? 'Done.'))
   185→                  : (action.response.isNotEmpty
   186→                        ? '${action.response}\n\n⚠️ ${result.details}'
   187→                        : '⚠️ ${result.details}'),
   188→              actionResult: result,
   189→            ),
   190→          );
   191→        });
   192→        _sendOverlayEvent(
   193→          'OVERLAY_TASK_FINISHED',
   194→          result.success
   195→              ? (result.details ?? 'Task complete.')
   196→              : 'Task failed: ${result.details ?? 'Unknown error'}',
   197→        );
   198→        if (action.action != 'execute_task') {
   199→          await _notificationService.showTaskCompleteNotification(
   200→            result.success ? 'Task Completed' : 'Task Failed',
   201→            result.details ??
   202→                (result.success
   203→                    ? 'Agent finished its goal.'
   204→                    : 'Agent could not complete the task.'),
   205→          );
   206→        }
   207→        await _saveSession();
   208→      } else {
   209→        // Plain text response, we already rendered it, just speak it
   210→        _voiceService.speak(accumulated);
   211→      }
   212→    } catch (e) {
   213→      if (mounted) {
   214→        setState(() {
   215→          if (_messages.isNotEmpty && _messages.length > assistantIndex) {
   216→            _messages.removeAt(assistantIndex);
   217→          }
   218→          _messages.add(
   219→            ChatMessage(
   220→              role: 'assistant',
   221→              content: 'Error: ${e.toString().replaceFirst('Exception: ', '')}',
   222→            ),
   223→          );
   224→        });
   225→      }
   226→    } finally {
   227→      if (mounted) {
   228→        setState(() {
   229→          _isLoading = false;
   230→        });
   231→        _scrollToBottom();
   232→        _updateOverlayState();
   233→      }
   234→    }
   235→  }

   237→  Future<void> _showTaskProgressOverlay(String message) async {
   238→    if (!FeatureFlags.floatingOverlayEnabled) return;
   239→    if (!await FlutterOverlayWindow.isPermissionGranted()) return;
   240→
   241→    // Never cover PrivateAgent itself. The lifecycle observer will create the
   242→    // overlay after an automated action moves this app to the background.
   243→    if (_appLifecycleState != AppLifecycleState.paused) return;
   244→
   245→    if (!await FlutterOverlayWindow.isActive()) {
   246→      await FlutterOverlayWindow.showOverlay(
   247→        enableDrag: true,
   248→        overlayTitle: 'PrivateAgent',
   249→        overlayContent: 'Performing task...',
   250→        flag: OverlayFlag.focusPointer,
   251→        alignment: OverlayAlignment.centerRight,
   252→        visibility: NotificationVisibility.visibilitySecret,
   253→        positionGravity: PositionGravity.auto,
   254→        startPosition: const OverlayPosition(0, 200),
   255→        width: 56,
   256→        height: 56,
   257→      );
   258→      await Future<void>.delayed(const Duration(milliseconds: 300));
   259→    }
   260→
   261→    // Keep the overlay minimized during automation. The user can still tap the
   262→    // bubble to open the full conversation whenever they choose.
   263→    _sendOverlayEvent('OVERLAY_TASK_STARTED', message);
   264→  }

   266→  void _sendOverlayEvent(String type, String message) {
   267→    if (!FeatureFlags.floatingOverlayEnabled) return;
   268→    final safeMessage = message.replaceAll('|', ' ');
   269→    unawaited(
   270→      FlutterOverlayWindow.shareData(
   271→        '$type|$safeMessage',
   272→      ).timeout(const Duration(seconds: 2)).catchError((Object _) {}),
   273→    );
   274→  }

   276→  Future<void> _sendOverlayHistorySnapshot() async {
   277→    if (!FeatureFlags.floatingOverlayEnabled) return;
   278→    final history = base64Encode(
   279→      utf8.encode(
   280→        jsonEncode(_messages.map((message) => message.toJson()).toList()),
   281→      ),
   282→    );
   283→    try {
   284→      await FlutterOverlayWindow.shareData(
   285→        'OVERLAY_HISTORY|$history',
   286→      ).timeout(const Duration(seconds: 2));
   287→    } catch (_) {}
   288→  }

   303→    if (_isListening) {
   304→      await _voiceService.stopListening();

   317→          setState(() => _isListening = false);

   325→      _sessionId = DateTime.now().millisecondsSinceEpoch.toString();

   327→      _messages.clear();
   328→      _aiService.clearHistory();

   332→  void _loadChatSession(ChatSession session) {
   333→    setState(() {
   334→      _sessionId = session.id;
   335→      _sessionTitle = session.title;
   336→      _messages.clear();
   337→      for (final m in session.messages) {
   338→        _messages.add(ChatMessage.fromJson(m));
   339→      }
   340→
   341→      _aiService.clearHistory();
   342→      for (final m in _messages) {
   343→        if (m.actionResult != null) continue;
   344→        _aiService.addHistoryMessage(m.role, m.content);
   345→      }
   346→    });
   347→    _scrollToBottom();
   348→  }

   361→  @override
   362→  void didChangeAppLifecycleState(AppLifecycleState state) {
   363→    setState(() {
   364→      _appLifecycleState = state;
   365→    });
   366→    if (state == AppLifecycleState.resumed) {
   367→      _startOverlayHistorySync();
   368→      unawaited(_handleAppForegrounded());
   369→    } else {
   370→      _overlayHistoryTimer?.cancel();
   371→      _updateOverlayState();
   372→    }
   373→  }
   374→
   375→  void _startOverlayHistorySync() {
   376→    _overlayHistoryTimer?.cancel();
   377→    if (!FeatureFlags.floatingOverlayEnabled) return;
   378→    unawaited(_importOverlayChatHistory());
   379→    _overlayHistoryTimer = Timer.periodic(const Duration(milliseconds: 500), (
   380→      _,
   381→    ) {
   382→      if (_appLifecycleState == AppLifecycleState.resumed) {
   383→        unawaited(_importOverlayChatHistory());
   384→      }
   385→    });
   386→  }
   387→
   388→  Future<void> _handleAppForegrounded() async {
   389→    await _updateOverlayState();
   390→    await _importOverlayChatHistory();
   391→  }
   392→
   393→  Future<void> _importOverlayChatHistory() async {
   394→    if (!FeatureFlags.floatingOverlayEnabled) return;
   395→    if (_importingOverlayHistory) return;
   396→    _importingOverlayHistory = true;
   397→    try {
   398→      final handoff = await ChatHistoryService.consumeOverlayMessages();
   399→      if (!mounted || handoff.isEmpty) return;
   400→
   401→      final imported = handoff.map(ChatMessage.fromJson).toList();
   402→      for (final message in imported) {
   403→        if (message.actionResult == null) {
   404→          _aiService.addHistoryMessage(message.role, message.content);
   405→        }
   406→      }
   407→      setState(() {
   408→        _messages.addAll(imported);
   409→      });
   410→      _scrollToBottom();
   411→      await _saveSession();
   412→    } finally {
   413→      _importingOverlayHistory = false;
   414→    }
   415→  }

   417→  int _overlayUpdateGeneration = 0;
   418→  bool _importingOverlayHistory = false;
   419→
   420→  Future<void> _updateOverlayState() async {
   421→    if (!FeatureFlags.floatingOverlayEnabled) return;
   422→    final generation = ++_overlayUpdateGeneration;
   423→    final isBackground = _appLifecycleState == AppLifecycleState.paused;
   424→    final shouldBeActive = isBackground;
   425→
   426→    bool granted = await FlutterOverlayWindow.isPermissionGranted();
   427→    if (!granted || generation != _overlayUpdateGeneration) return;
   428→
   429→    bool active = await FlutterOverlayWindow.isActive();
   430→    if (generation != _overlayUpdateGeneration) return;
   431→    if (shouldBeActive && !active) {
   432→      await Future.delayed(const Duration(milliseconds: 200));
   433→      if (generation != _overlayUpdateGeneration) return;
   434→      if (_appLifecycleState != AppLifecycleState.paused) return;
   435→      if (await FlutterOverlayWindow.isActive()) return;
   436→      await FlutterOverlayWindow.showOverlay(
   437→        enableDrag: true,
   438→        overlayTitle: "PrivateAgent",
   439→        overlayContent: _isLoading
   440→            ? "Performing task..."
   441→            : "Floating Assistant",
   442→        flag: OverlayFlag.focusPointer,
   443→        alignment: OverlayAlignment.centerRight,
   444→        visibility: NotificationVisibility.visibilitySecret,
   445→        positionGravity: PositionGravity.auto,
   446→        startPosition: const OverlayPosition(0, 200),
   447→        width: 56,
   448→        height: 56,
   449→      );
   450→      if (_isLoading && _appLifecycleState == AppLifecycleState.paused) {
   451→        // Give the overlay isolate time to attach its listener, then send the
   452→        // full active conversation. A second snapshot makes cold starts
   453→        // reliable without duplicating messages because the overlay replaces
   454→        // its list atomically.
   455→        await Future<void>.delayed(const Duration(milliseconds: 250));
   456→        await _sendOverlayHistorySnapshot();
   457→        await Future<void>.delayed(const Duration(milliseconds: 250));
   458→        if (_isLoading && _appLifecycleState == AppLifecycleState.paused) {
   459→          await _sendOverlayHistorySnapshot();
   460→        }
   461→      }
   462→    } else if (shouldBeActive && active && _isLoading) {
   463→      await _sendOverlayHistorySnapshot();
   464→    } else if (!shouldBeActive && active) {
   465→      try {
   466→        await FlutterOverlayWindow.shareData(
   467→          'OVERLAY_RESET|',
   468→        ).timeout(const Duration(milliseconds: 150));
   469→      } catch (_) {}
   470→      await Future<void>.delayed(const Duration(milliseconds: 50));
   471→      if (generation != _overlayUpdateGeneration) return;
   472→      if (_appLifecycleState == AppLifecycleState.paused) return;
   473→      await FlutterOverlayWindow.closeOverlay();
   474→    }
   475→  }
