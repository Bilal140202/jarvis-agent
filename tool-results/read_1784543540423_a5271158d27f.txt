    84→  Future<void> _checkOverlayStatus() async {
    85→    bool isActive = await FlutterOverlayWindow.isActive();
    86→    bool isGranted = await FlutterOverlayWindow.isPermissionGranted();
    87→    if (mounted) {
    88→      setState(() {
    89→        _floatingIconEnabled = isActive;
    90→        _isOverlayPermissionGranted = isGranted;
    91→      });
    92→    }
    93→  }
    94→
    95→  @override
    96→  void dispose() {
    97→    WidgetsBinding.instance.removeObserver(this);
    98→    _apiKeyController.removeListener(_autoSave);
    99→    _baseUrlController.removeListener(_autoSave);
   100→    _modelController.removeListener(_autoSave);
   101→    _telegramTokenController.removeListener(_autoSave);
   102→    _maxTokensController.removeListener(_autoSave);
   103→    _apiKeyController.dispose();
   104→    _baseUrlController.dispose();
   105→    _modelController.dispose();
   106→    _telegramTokenController.dispose();
   107→    _maxTokensController.dispose();
   108→    super.dispose();
   109→  }
   110→
   111→  @override
   112→  void didChangeAppLifecycleState(AppLifecycleState state) {
   113→    if (state == AppLifecycleState.resumed) {
   114→      _checkPermissions();
   115→      if (FeatureFlags.floatingOverlayEnabled) {
   116→        _checkOverlayStatus();
   117→      }
   118→    }
   119→  }
   120→
   121→  Future<void> _checkPermissions() async {
   122→    final perms = {
   123→      'Microphone': Permission.microphone,
   124→      'Contacts': Permission.contacts,
   125→      'Phone': Permission.phone,
   126→      'SMS': Permission.sms,
   127→      'Notifications': Permission.notification,
   128→    };
   129→
   130→    for (final entry in perms.entries) {
   131→      _permissions[entry.key] = await entry.value.status;
   132→    }
   133→    final overlayGranted = FeatureFlags.floatingOverlayEnabled
   134→        ? await FlutterOverlayWindow.isPermissionGranted()
   135→        : false;
   136→    if (mounted) {
   137→      setState(() {
   138→        _isOverlayPermissionGranted = overlayGranted;
   139→      });
   140→    }
   141→  }
   142→
   143→  Future<void> _requestPermission(String name, Permission permission) async {
   144→    final status = await permission.request();
   145→    setState(() => _permissions[name] = status);
   146→  }
   147→
   148→  void _autoSave() {
   149→    widget.aiService.saveSettings(
   150→      apiKey: _apiKeyController.text.trim(),
   151→      baseUrl: _baseUrlController.text.trim(),
   152→      model: _modelController.text.trim(),
   153→    );
   154→
   155→    widget.telegramService.saveSettings(
   156→      botToken: _telegramTokenController.text.trim(),
   157→      isEnabled: _telegramEnabled,
   158→    );
   159→
   160→    widget.aiService.saveMaxSteps(_maxSteps.toInt());
   161→    widget.aiService.saveDisableMaxSteps(_disableMaxSteps);
   162→    widget.aiService.saveAdvancedSettings(
   163→      temperature: _temperature,
   164→      maxTokens: int.tryParse(_maxTokensController.text) ?? 1024,
   165→      useScreenCompression: _useScreenCompression,
   166→      useSystemPrompt: _useSystemPrompt,
   167→    );
   168→  }

   170→  Future<void> _fetchModels() async {
   171→    final baseUrl = _baseUrlController.text.trim();
   172→    final apiKey = _apiKeyController.text.trim();
   173→
   174→    if (baseUrl.isEmpty || apiKey.isEmpty) {
   175→      ScaffoldMessenger.of(context).showSnackBar(
   176→        const SnackBar(
   177→          content: Text('Please enter Base URL and API Key first.'),
   178→        ),
   179→      );
   180→      return;
   181→    }
   182→
   183→    // Show loading
   184→    showDialog(
   185→      context: context,
   186→      barrierDismissible: false,
   187→      builder: (context) => const Center(child: CircularProgressIndicator()),
   188→    );
   189→
   190→    final models = await widget.aiService.fetchAvailableModels(baseUrl, apiKey);
   191→
   192→    // Hide loading
   193→    if (mounted) Navigator.pop(context);
   194→
   195→    if (models.isEmpty) {
   196→      if (mounted) {
   197→        ScaffoldMessenger.of(context).showSnackBar(
   198→          const SnackBar(
   199→            content: Text('No models found or error fetching models.'),
   200→          ),
   201→        );
   202→      }
   203→      return;
   204→    }
   205→
   206→    if (mounted) {
   207→      final isNvidia = AiService.isNvidiaBaseUrl(baseUrl);
   208→      showDialog(
   209→        context: context,
   210→        builder: (context) => AlertDialog(
   211→          title: Text(
   212→            isNvidia ? 'Select a Free NVIDIA Model' : 'Select a Model',
   213→          ),
   214→          content: SizedBox(
   215→            width: double.maxFinite,
   216→            height: 300,
   217→            child: ListView.builder(
   218→              itemCount: models.length,
   219→              itemBuilder: (context, index) {
   220→                return ListTile(
   221→                  title: Text(models[index]),
   222→                  onTap: () {
   223→                    setState(() {
   224→                      _modelController.text = models[index];
   225→                    });
   226→                    Navigator.pop(context);
   227→                  },
   228→                );
   229→              },
   230→            ),
   231→          ),
   232→          actions: [
   233→            TextButton(
   234→              onPressed: () => Navigator.pop(context),
   235→              child: const Text('Cancel'),
   236→            ),
   237→          ],
   238→        ),
   239→      );
   240→    }
   241→  }

   881→  List<Widget> _buildPermissionTiles() {
   882→    final permissionMap = {
   883→      'Microphone': Permission.microphone,
   884→      'Contacts': Permission.contacts,
   885→      'Phone': Permission.phone,
   886→      'SMS': Permission.sms,
   887→      'Notifications': Permission.notification,
   888→    };
   889→
   890→    final icons = {
   891→      'Microphone': Icons.mic,
   892→      'Contacts': Icons.contacts,
   893→      'Phone': Icons.phone,
   894→      'SMS': Icons.sms,
   895→      'Notifications': Icons.notifications,
   896→    };
   897→
   898→    final list = permissionMap.entries.map((entry) {
   899→      final status = _permissions[entry.key];
   900→      final isGranted = status?.isGranted ?? false;
   901→
   902→      return ListTile(
   903→        leading: Icon(icons[entry.key]),
   904→        title: Text(entry.key),
   905→        trailing: isGranted
   906→            ? Icon(
   907→                Icons.check_circle,
   908→                color: Theme.of(context).colorScheme.primary,
   909→              )
   910→            : TextButton(
   911→                onPressed: () => _requestPermission(entry.key, entry.value),
   912→                child: const Text('Grant'),
   913→              ),
   914→        subtitle: Text(
   915→          isGranted
   916→              ? 'Granted'
   917→              : (status?.isDenied ?? true
   918→                    ? 'Not granted'
   919→                    : 'Denied permanently'),
   920→          style: TextStyle(
   921→            color: isGranted
   922→                ? Theme.of(context).colorScheme.primary
   923→                : Colors.orange,
   924→            fontSize: 12,
   925→          ),
   926→        ),
   927→      );
   928→    }).toList();
   929→
   930→    if (FeatureFlags.floatingOverlayEnabled) {
   931→      list.add(
   932→        ListTile(
   933→          leading: const Icon(Icons.layers),
   934→          title: const Text('Display Over Other Apps (Floating Bubble)'),
   935→          trailing: _isOverlayPermissionGranted
   936→              ? Icon(
   937→                  Icons.check_circle,
   938→                  color: Theme.of(context).colorScheme.primary,
   939→                )
   940→              : TextButton(
   941→                  onPressed: () async {
   942→                    await FlutterOverlayWindow.requestPermission();
   943→                    final granted =
   944→                        await FlutterOverlayWindow.isPermissionGranted();
   945→                    setState(() {
   946→                      _isOverlayPermissionGranted = granted;
   947→                    });
   948→                  },
   949→                  child: const Text('Grant'),
   950→                ),
   951→          subtitle: Text(
   952→            _isOverlayPermissionGranted ? 'Granted' : 'Not granted',
   953→            style: TextStyle(
   954→              color: _isOverlayPermissionGranted
   955→                  ? Theme.of(context).colorScheme.primary
   956→                  : Colors.orange,
   957→              fontSize: 12,
   958→            ),
   959→          ),
   960→        ),
   961→      );
   962→    }
   963→
   964→    return list;
   965→  }
